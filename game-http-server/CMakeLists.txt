cmake_minimum_required(VERSION 3.11)
project(game_server CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(THREADS_PREFER_PTHREAD_FLAG ON)
include(${CMAKE_BINARY_DIR}/conanbuildinfo.cmake)
conan_basic_setup(TARGETS)
find_package(Threads REQUIRED)
set(COMMON src/model.cpp src/model.h src/tagged.h src/loot_generator.cpp src/loot_generator.h src/collision_detector.cpp src/collision_detector.h src/geom.h src/model_serialization.cpp src/model_serialization.h)
add_executable(game_server src/main.cpp src/http_server.cpp src/http_server.h src/server_logging.h src/sdk.h src/boost_json.cpp src/json_loader.cpp src/json_loader.h src/extra_data.cpp src/extra_data.h src/request_handler.cpp src/request_handler.h src/application.h src/ticker.cpp src/ticker.h src/state_serializer.cpp src/state_serializer.h src/records_repository.cpp src/records_repository.h ${COMMON})
target_link_libraries(game_server PRIVATE CONAN_PKG::boost CONAN_PKG::libpqxx Threads::Threads)
add_executable(game_server_tests tests/loot_generator_tests.cpp tests/model_tests.cpp tests/state-serialization-tests.cpp ${COMMON})
target_link_libraries(game_server_tests PRIVATE CONAN_PKG::catch2 CONAN_PKG::boost Threads::Threads)
if(WIN32)
 target_link_libraries(game_server PRIVATE ws2_32 mswsock)
 target_link_libraries(game_server_tests PRIVATE ws2_32 mswsock)
endif()
